一、测试硬件
服务器：4xL40s（默认ECC打开情况下，显存为46GB）

二、建立可复现的OPCD基线
1、下载OPCD(/home/zs/notebooks/LMOps)
    git clone https://github.com/microsoft/LMOps

2、安装环境
    1）python3.12的venv
        python3.12 -m venv ~/venvs/opcd_verl
        source ~/venvs/opcd_verl/bin/activate
        cd LMOps/opcd
    2）安装基础工具
        source ~/venvs/opcd_verl/bin/activate
        python -m pip install --upgrade \
            pip \
            wheel \
            "setuptools>=77,<80" \
            packaging \
            ninja \
            psutil \
            uv
    3）安装PyTorch2.7、vLLM 0.9和基础依赖(暂时不安装 trl 和 textarena)
        source ~/venvs/opcd_verl/bin/activate
        echo "$VIRTUAL_ENV"
        uv pip install \
            --python  "$VIRTUAL_ENV/bin/python" \
            --torch-backend=cu128 \
            "vllm==0.9.0" \
            "transformers==4.52.3" \
            "ray[default]==2.43.0" \
            "datasets==4.0.0" \
            "tensordict==0.6.2" \
            "torchdata==0.11.0" \
            "hydra-core==1.3.2" \
            "wandb==0.19.7" \
            "peft==0.15.2" \
            "rouge-score"
    4）编译 FlashAttention-2(2.8.3.post1为目前最新版本，并不一定必须这个版本，只是固定一下比较好；另外，FlashAttention-3/4  不适合 L40S)
        export CUDA_HOME=/usr/local/cuda-12.8
        export PATH="${CUDA_HOME}/bin:${VIRTUAL_ENV}/bin:${PATH}"
        export LD_LIBRARY_PATH="${CUDA_HOME}/lib64"
        export TORCH_CUDA_ARCH_LIST=8.9
        export MAX_JOBS=8

        python -m pip install \
            "flash-attn==2.8.3.post1" \
            --no-build-isolation \
            --no-cache-dir
    5）安装opcd中的verl
        cd /home/zs/notebooks/LMOps/opcd
        source /home/zs/venvs/opcd_verl/bin/activate
        python -m pip install -e /home/zs/notebooks/LMOps/opcd/verl --no-deps
