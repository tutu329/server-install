        export CUDA_VISIBLE_DEVICES=0,1,2,3

        python - <<'PY'
        import torch
        from flash_attn import flash_attn_func

        for i in range(torch.cuda.device_count()):
            device = f"cuda:{i}"

            q = torch.randn(
                1, 256, 8, 64,
                device=device,
                dtype=torch.bfloat16,
                requires_grad=True,
            )
            k = torch.randn_like(q, requires_grad=True)
            v = torch.randn_like(q, requires_grad=True)

            out = flash_attn_func(
                q, k, v,
                dropout_p=0.0,
                causal=True,
            )
            loss = out.float().square().mean()
            loss.backward()
            torch.cuda.synchronize(i)

            print(
                f"GPU {i}: {torch.cuda.get_device_name(i)}, "
                f"output={tuple(out.shape)}, "
                f"loss={loss.item():.6f}, "
                f"forward/backward=OK"
            )

        print("4-GPU FlashAttention smoke test passed")
        PY

        from transformers import AutoTokenizer
        from vllm import LLM, SamplingParams

        model_path = "/home/zs/models/Qwen2.5-0.5B-Instruct"

        tokenizer = AutoTokenizer.from_pretrained(
            model_path,
            trust_remote_code=False,
        )

        messages = [
            {
                "role": "user",
                "content": "请只回答：OPCD M0 vLLM 测试成功",
            }
        ]

        prompt = tokenizer.apply_chat_template(
            messages,
            tokenize=False,
            add_generation_prompt=True,
        )

        llm = LLM(
            model=model_path,
            tensor_parallel_size=1,
            dtype="bfloat16",
            max_model_len=1024,
            gpu_memory_utilization=0.20,
            enforce_eager=True,
            trust_remote_code=False,
        )

        sampling_params = SamplingParams(
            temperature=0.0,
            max_tokens=32,
        )

        outputs = llm.generate([prompt], sampling_params)
        text = outputs[0].outputs[0].text

        print("\n===== GENERATED TEXT =====")
        print(text)
        print("==========================")
        print("Single-GPU local vLLM inference = OK")
        PY

1）NCCL测试
    # 生成py
        tee /tmp/opcd_nccl_smoke.py >/dev/null <<'PY'
        import os
        from datetime import timedelta

        import torch
        import torch.distributed as dist

        rank = int(os.environ["RANK"])
        local_rank = int(os.environ["LOCAL_RANK"])
        world_size = int(os.environ["WORLD_SIZE"])

        torch.cuda.set_device(local_rank)
        device = torch.device("cuda", local_rank)

        print(
            f"rank={rank}/{world_size}, "
            f"local_rank={local_rank}, "
            f"device={torch.cuda.get_device_name(local_rank)}, "
            f"CUDA_VISIBLE_DEVICES={os.environ.get('CUDA_VISIBLE_DEVICES')}",
            flush=True,
        )

        dist.init_process_group(
            backend="nccl",
            timeout=timedelta(seconds=90),
        )

        small = torch.tensor([float(rank + 1)], device=device)
        dist.all_reduce(small)
        torch.cuda.synchronize()

        large = torch.full(
            (14_912_384,),
            -1.0,
            dtype=torch.float32,
            device=device,
        )

        if rank == 0:
            large.fill_(7.0)

        dist.broadcast(large, src=0)
        torch.cuda.synchronize()

        assert small.item() == 10.0, small.item()
        assert large[0].item() == 7.0, large[0].item()
        assert large[-1].item() == 7.0, large[-1].item()

        print(
            f"rank={rank}: all_reduce={small.item()}, "
            f"broadcast={large[0].item()}, NCCL OK",
            flush=True,
        )

        dist.barrier()
        dist.destroy_process_group()
        PY

2）NCCL测试失败：
    日志结果表明：
        NCCL communicator 初始化成功；
        GPU 0↔1、GPU 2↔3 使用 P2P/CUMEM；
        两组 GPU 之间使用 SHM；
        rank 0 完成了 collective，但 rank 1/2/3 卡在最小的 ALLREDUCE NumelIn=1；
        90 秒后超时。
        这高度指向当前机器的 PCIe P2P/CUMEM/共享内存通信路径异常。
    保守测试：
        export NCCL_CUMEM_HOST_ENABLE=0
        export NCCL_P2P_DISABLE=1
        export NCCL_IB_DISABLE=1
        export NCCL_SHM_DISABLE=0

        export NCCL_DEBUG=INFO
        export NCCL_DEBUG_SUBSYS=INIT,GRAPH,NET
        export TORCH_NCCL_ASYNC_ERROR_HANDLING=1

        set -o pipefail

        timeout --signal=TERM --kill-after=10s 150s \
          python -m torch.distributed.run \
            --standalone \
            --nproc_per_node=4 \
            /tmp/opcd_nccl_smoke.py \
